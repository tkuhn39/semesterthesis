% Die Gruppen des Literaturverzeichnisses wie in FZGdef.sty, nur mit gültigen Filtern: dort
% stehen nottype={manual,online} und notkeyword={norm,web} als Listen, die biblatex nicht
% kennt, so dass Patente (keyword norm) unter den Büchern und Internetseiten doppelt erschienen.
\renewcommand{\insertbib}{%
    \printbibliography[type=manual,title={\titleNormen},heading=subbibliography]
    \printbibliography[nottype=manual,keyword={norm},heading=none]
    \printbibliography[nottype=manual,nottype=online,notkeyword={norm},notkeyword={web},title={\titleJournals},heading=subbibliography]
    \printbibliography[type=online,title={\titleURLS},heading=subbibliography]
    \printbibliography[nottype=manual,nottype=online,keyword={web},heading=none]
}
